### 5. server.js

```javascript
const express = require('express');
const http = require('http');
const socketIO = require('socket.io');
const cors = require('cors');
const dotenv = require('dotenv');
const connectDB = require('./config/database');
const { logger } = require('./src/utils/logger');
const WhatsAppClient = require('./src/whatsapp/client');
const MessageHandler = require('./src/whatsapp/messageHandler');
const TaskScheduler = require('./src/tasks/scheduler');
const dashboardRoutes = require('./src/dashboard/routes');

// Load environment variables
dotenv.config();

// Initialize Express
const app = express();
const server = http.createServer(app);
const io = socketIO(server, {
  cors: {
    origin: "*",
    methods: ["GET", "POST"]
  }
});

// Middleware
app.use(cors());
app.use(express.json());
app.use(express.urlencoded({ extended: true }));
app.use(express.static('src/dashboard/views'));

// Connect to Database
connectDB();

// Global state
let whatsappClient = null;
let messageHandler = null;

// Initialize WhatsApp Client
async function initializeWhatsApp() {
  try {
    logger.info('Initializing WhatsApp client...');
    
    whatsappClient = new WhatsAppClient();
    messageHandler = new MessageHandler(whatsappClient, io);
    
    await whatsappClient.initialize();
    
    // Emit events to dashboard
    whatsappClient.on('qr', (qr) => {
      io.emit('qr', qr);
    });
    
    whatsappClient.on('ready', () => {
      logger.info('WhatsApp client is ready!');
      io.emit('status', { connected: true });
    });
    
    whatsappClient.on('message', async (msg) => {
      await messageHandler.handle(msg);
    });
    
    whatsappClient.on('disconnected', () => {
      logger.warn('WhatsApp client disconnected');
      io.emit('status', { connected: false });
    });
    
  } catch (error) {
    logger.error('Failed to initialize WhatsApp:', error);
    throw error;
  }
}

// Initialize Task Scheduler
const taskScheduler = new TaskScheduler(whatsappClient);
taskScheduler.start();

// Routes
app.use('/api', dashboardRoutes);

app.get('/', (req, res) => {
  res.sendFile(__dirname + '/src/dashboard/views/index.html');
});

app.get('/health', (req, res) => {
  res.json({ 
    status: 'ok', 
    whatsapp: whatsappClient?.isConnected() || false,
    uptime: process.uptime()
  });
});

// Socket.IO for dashboard
io.on('connection', (socket) => {
  logger.info('Dashboard client connected');
  
  socket.on('take-control', (data) => {
    messageHandler.takeControl(data.contactId);
  });
  
  socket.on('release-control', (data) => {
    messageHandler.releaseControl(data.contactId);
  });
  
  socket.on('send-message', async (data) => {
    await whatsappClient.sendMessage(data.to, data.message);
  });
});

// Error handling
process.on('unhandledRejection', (error) => {
  logger.error('Unhandled rejection:', error);
});

process.on('uncaughtException', (error) => {
  logger.error('Uncaught exception:', error);
  process.exit(1);
});

// Start server
const PORT = process.env.PORT || 3000;

server.listen(PORT, async () => {
  logger.info(`Server running on port ${PORT}`);
  logger.info(`Dashboard available at http://localhost:${PORT}`);
  
  // Initialize WhatsApp
  await initializeWhatsApp();
});

// Graceful shutdown
process.on('SIGINT', async () => {
  logger.info('Shutting down gracefully...');
  
  if (whatsappClient) {
    await whatsappClient.disconnect();
  }
  
  process.exit(0);
});

module.exports = { app, whatsappClient };
```

### 7. config/llm.js

```javascript
const OpenAI = require('openai');
const axios = require('axios');
const { logger } = require('../src/utils/logger');

class LLMConfig {
  constructor() {
    this.provider = process.env.LLM_PROVIDER || 'openai';
    this.initializeClient();
  }
  
  initializeClient() {
    switch(this.provider) {
      case 'openai':
        this.client = new OpenAI({
          apiKey: process.env.OPENAI_API_KEY
        });
        this.model = process.env.OPENAI_MODEL || 'gpt-4o-mini';
        break;
        
      case 'anthropic':
        this.apiKey = process.env.ANTHROPIC_API_KEY;
        this.model = process.env.ANTHROPIC_MODEL || 'claude-3-haiku-20240307';
        break;
        
      case 'huggingface':
        this.apiKey = process.env.HUGGINGFACE_API_KEY;
        this.model = process.env.HUGGINGFACE_MODEL || 'meta-llama/Llama-3.2-3B-Instruct';
        break;
        
      case 'together':
        this.apiKey = process.env.TOGETHER_API_KEY;
        this.model = process.env.TOGETHER_MODEL || 'mistralai/Mixtral-8x7B-Instruct-v0.1';
        break;
        
      default:
        throw new Error(`Unknown LLM provider: ${this.provider}`);
    }
  }
  
  async generateCompletion(messages, options = {}) {
    try {
      switch(this.provider) {
        case 'openai':
          return await this.openAICompletion(messages, options);
          
        case 'anthropic':
          return await this.anthropicCompletion(messages, options);
          
        case 'huggingface':
          return await this.huggingFaceCompletion(messages, options);
          
        case 'together':
          return await this.togetherCompletion(messages, options);
      }
    } catch (error) {
      logger.error(`LLM completion failed (${this.provider}):`, error);
      throw error;
    }
  }
  
  async openAICompletion(messages, options) {
    const response = await this.client.chat.completions.create({
      model: this.model,
      messages: messages,
      temperature: options.temperature || 0.7,
      max_tokens: options.maxTokens || 500
    });
    
    return response.choices[0].message.content;
  }
  
  async anthropicCompletion(messages, options) {
    const response = await axios.post(
      'https://api.anthropic.com/v1/messages',
      {
        model: this.model,
        messages: messages,
        max_tokens: options.maxTokens || 500
      },
      {
        headers: {
          'x-api-key': this.apiKey,
          'anthropic-version': '2023-06-01',
          'content-type': 'application/json'
        }
      }
    );
    
    return response.data.content[0].text;
  }
  
  async huggingFaceCompletion(messages, options) {
    // Convert messages to single prompt
    const prompt = messages.map(m => `${m.role}: ${m.content}`).join('\n');
    
    const response = await axios.post(
      `https://api-inference.huggingface.co/models/${this.model}`,
      {
        inputs: prompt,
        parameters: {
          max_new_tokens: options.maxTokens || 500,
          temperature: options.temperature || 0.7
        }
      },
      {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`
        }
      }
    );
    
    return response.data[0].generated_text;
  }
  
  async togetherCompletion(messages, options) {
    const response = await axios.post(
      'https://api.together.xyz/v1/chat/completions',
      {
        model: this.model,
        messages: messages,
        max_tokens: options.maxTokens || 500,
        temperature: options.temperature || 0.7
      },
      {
        headers: {
          'Authorization': `Bearer ${this.apiKey}`,
          'Content-Type': 'application/json'
        }
      }
    );
    
    return response.data.choices[0].message.content;
  }
}

module.exports = new LLMConfig();
```

const { logger } = require('../utils/logger');
const fs = require('fs');
const path = require('path');

class WhatsAppClient extends EventEmitter {
  constructor() {
    super();
    this.sock = null;
    this.isReady = false;
    this.sessionPath = process.env.WHATSAPP_SESSION_PATH || './auth_info';
  }
  
  async initialize() {
    try {
      // Ensure session directory exists
      if (!fs.existsSync(this.sessionPath)) {
        fs.mkdirSync(this.sessionPath, { recursive: true });
      }
      
      const { state, saveCreds } = await useMultiFileAuthState(this.sessionPath);
      
      this.sock = makeWASocket({
        auth: state,
        printQRInTerminal: true,
        logger: logger
      });
      
      // Handle QR code
      this.sock.ev.on('connection.update', async (update) => {
        const { connection, lastDisconnect, qr } = update;
        
        if (qr) {
          logger.info('QR Code received');
          qrcode.generate(qr, { small: true });
          this.emit('qr', qr);
        }
        
        if (connection === 'close') {
          const shouldReconnect = lastDisconnect?.error?.output?.statusCode !== DisconnectReason.loggedOut;
          
          logger.warn('Connection closed', { shouldReconnect });
          
          if (shouldReconnect) {
            setTimeout(() => this.initialize(), 5000);
          }
          
          this.isReady = false;
          this.emit('disconnected');
          
        } else if (connection === 'open') {
          logger.info('WhatsApp connection established!');
          this.isReady = true;
          this.emit('ready');
        }
      });
      
      // Save credentials
      this.sock.ev.on('creds.update', saveCreds);
      
      // Handle messages
      this.sock.ev.on('messages.upsert', async ({ messages, type }) => {
        if (type === 'notify') {
          for (const msg of messages) {
            if (!msg.key.fromMe && msg.message) {
              this.emit('message', msg);
            }
          }
        }
      });
      
    } catch (error) {
      logger.error('Failed to initialize WhatsApp client:', error);
      throw error;
    }
  }
  
  async sendMessage(to, message) {
    try {
      if (!this.isReady) {
        throw new Error('WhatsApp client not ready');
      }
      
      // Ensure proper JID format
      const jid = to.includes('@') ? to : `${to}@s.whatsapp.net`;
      
      await this.sock.sendMessage(jid, { text: message });
      logger.info('Message sent', { to: jid });
      
      return true;
    } catch (error) {
      logger.error('Failed to send message:', error);
      throw error;
    }
  }
  
  async sendMedia(to, mediaBuffer, caption = '') {
    try {
      const jid = to.includes('@') ? to : `${to}@s.whatsapp.net`;
      
      await this.sock.sendMessage(jid, {
        image: mediaBuffer,
        caption: caption
      });
      
      return true;
    } catch (error) {
      logger.error('Failed to send media:', error);
      throw error;
    }
  }
  
  async disconnect() {
    if (this.sock) {
      await this.sock.logout();
      this.isReady = false;
    }
  }
  
  isConnected() {
    return this.isReady;
  }
}

module.exports = WhatsAppClient;

const Contact = require('../models/Contact');

class MessageHandler {
  constructor(whatsappClient, io) {
    this.client = whatsappClient;
    this.io = io;
    this.manualControlContacts = new Set();
    this.conversationController = new ConversationController();
    this.personalityEngine = new PersonalityEngine();
    this.memoryManager = new MemoryManager();
  }
  
  async handle(msg) {
    try {
      const from = msg.key.remoteJid;
      const phone = from.replace('@s.whatsapp.net', '');
      const messageContent = this.extractMessageContent(msg);
      
      logger.info('Message received', { from: phone, content: messageContent });
      
      // Get or create contact
      let contact = await Contact.findOne({ phone });
      if (!contact) {
        contact = await this.createNewContact(phone, msg);
      }
      
      // Check if manual control is active
      if (this.manualControlContacts.has(contact._id.toString())) {
        logger.info('Manual control active, forwarding to dashboard');
        this.io.emit('manual-message', {
          contactId: contact._id,
          contactName: contact.name,
          message: messageContent
        });
        return;
      }
      
      // Check if contact is paused
      if (contact.current_status.is_paused) {
        logger.info('Contact is paused, skipping');
        return;
      }
      
      // Log the message
      await this.conversationController.logMessage(
        contact._id,
        messageContent,
        'them',
        msg
      );
      
      // Generate AI response
      const response = await this.generateResponse(contact, messageContent);
      
      // Send response
      await this.client.sendMessage(from, response);
      
      // Log bot response
      await this.conversationController.logMessage(
        contact._id,
        response,
        'bot'
      );
      
      // Update contact last interaction
      contact.current_status.last_interaction = new Date();
      await contact.save();
      
      // Emit to dashboard
      this.io.emit('conversation-update', {
        contactId: contact._id,
        message: messageContent,
        response: response
      });
      
    } catch (error) {
      logger.error('Error handling message:', error);
    }
  }

    return msg.message?.conversation || 
           msg.message?.extendedTextMessage?.text || 
           '';
  }
  
  async createNewContact(phone, msg) {
    // Try to get name from WhatsApp
    const name = msg.pushName || phone;
    
    const contact = new Contact({
      phone,
      name,
      relationship_type: 'casual',
      current_status: {
        last_interaction: new Date()
      }
    });
    
    await contact.save();
    logger.info('New contact created', { phone, name });
    
    return contact;
  }
  
  async generateResponse(contact, message) {
    try {
      // Build context from memory
      const context = await this.memoryManager.buildContext(contact._id, message);
      
      // Generate response using personality engine
      const response = await this.personalityEngine.generateResponse(

        message,
        context
      );

      logger.error('Failed to generate response:', error);

    }
  }
  
  takeControl(contactId) {
    this.manualControlContacts.add(contactId);

  }
  
  releaseControl(contactId) {
    this.manualControlContacts.delete(contactId);
    logger.info('Manual control released', { contactId });
  }
}

const Memory = require('../models/Memory');

class MemoryManager {
  constructor() {
    this.cacheSize = 100; // Cache last 100 accessed memories
    this.cache = new Map();
  }
  
  async buildContext(contactId, currentMessage) {
    try {
      const context = {
        memories: [],
        recentMessages: [],
        activeTasks: []
      };
      
      // Get relevant memories (semantic search simulation)
      context.memories = await this.getRelevantMemories(contactId, currentMessage);
      
      // Get recent conversation history
      context.recentMessages = await this.getRecentMessages(contactId, 10);
      
      // Get active tasks for this contact
      context.activeTasks = await this.getActiveTasks(contactId);
      
      return context;
      
    } catch (error) {
      logger.error('Failed to build context:', error);
      return {};
    }
  }
  
  async getRelevantMemories(contactId, message, limit = 5) {
    try {
      // Simple keyword-based search (in production, use vector search)
      const keywords = message.toLowerCase().split(' ').filter(w => w.length > 3);
      
      const memories = await Memory.find({
        $or: [
          { related_contacts: contactId },
          { content: { $regex: keywords.join('|'), $options: 'i' } }
        ],
        $or: [
          { expires_at: null },
          { expires_at: { $gt: new Date() } }
        ]
      })
      .sort({ priority: -1, access_count: -1 })
      .limit(limit);
      
      // Update access count
      for (const mem of memories) {
        mem.access_count += 1;
        mem.last_accessed = new Date();
        await mem.save();
      }
      
      return memories;
      
    } catch (error) {
      logger.error('Failed to get relevant memories:', error);
      return [];
    }
  }
  
  async getRecentMessages(contactId, limit = 10) {
    try {
      const conversations = await Conversation.find({
        contact_id: contactId,
        is_active: true
      })
      .sort({ start_time: -1 })
      .limit(1);
      
      if (conversations.length === 0) return [];
      
      const recentConv = conversations[0];
      return recentConv.messages.slice(-limit);
      
    } catch (error) {
      logger.error('Failed to get recent messages:', error);
      return [];
    }
  }
  
  async getActiveTasks(contactId) {
    try {
      return await Task.find({
        contact_id: contactId,
        status: { $in: ['pending', 'in_progress'] }
      })
      .sort({ priority: -1, deadline: 1 })
      .limit(5);
      
    } catch (error) {
      logger.error('Failed to get active tasks:', error);
      return [];
    }
  }
  
  async storeMemory(content, category, priority, contactIds = [], topics = []) {
    try {
      const memory = new Memory({
        content,
        category,
        priority,
        related_contacts: contactIds,
        related_topics: topics

const Conversation = require('../models/Conversation');

class ConversationController {
  async logMessage(contactId, content, sender, rawMsg = null) {
    try {
      // Find or create active conversation
      let conversation = await Conversation.findOne({
        contact_id: contactId,
        is_active: true
      });
      
      if (!conversation) {
        conversation = new Conversation({
          contact_id: contactId,
          messages: []
        });
      }
      
      // Add message
      conversation.messages.push({
        msg_id: rawMsg?.key?.id || Date.now().toString(),
        timestamp: new Date(),
        sender,
        content,
        sentiment: this.detectSentiment(content),
        intent: this.detectIntent(content)
      });
      
      await conversation.save();
      
      return conversation;
      
    } catch (error) {
      logger.error('Failed to log message:', error);
      throw error;
    }
  }
  
  detectSentiment(content) {
    // Simple sentiment detection
    const positive = ['good', 'great', 'awesome', 'thanks', 'love', '😊', '👍', '❤️'];
    const negative = ['bad', 'hate', 'angry', 'sad', '😞', '😡'];
    
    const lowerContent = content.toLowerCase();
    
    if (positive.some(word => lowerContent.includes(word))) {
      return 'positive';
    }
    if (negative.some(word => lowerContent.includes(word))) {
      return 'negative';
    }
    
    return 'neutral';
  }
  
  detectIntent(content) {
    if (content.includes('?')) return 'question';
    if (content.toLowerCase().includes('please') || content.toLowerCase().includes('can you')) {
      return 'request';
    }
    
    return 'statement';
  }
}

module.exports = ConversationController;

const path = require('path');

const structure = {

      fs.mkdirSync(dir, { recursive: true });
